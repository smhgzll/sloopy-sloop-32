# The network and the web panel. The panel's files (web/*.html, *.js, *.css) are gzipped at build
# time and embedded in the firmware; web_assets.c (generated) lists them for the HTTP server.
set(WEB_DIR "${COMPONENT_DIR}/../../web")
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/web_assets.c")

idf_component_register(
    SRCS "net.c" "web_server.c" "settings.c" "ota.c" "backup.c" "${GEN}"
    INCLUDE_DIRS "include"
    PRIV_INCLUDE_DIRS "."
    REQUIRES esp_http_server esp_netif esp_event
    PRIV_REQUIRES sloop_proto sloop_core sloop_port esp_wifi esp_eth log espressif__mdns nvs_flash app_update
                  esp_app_format bootloader_support
)

if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    # the panel (web/) and SLOOP's web editor (upstream, + the bridge line; cmake/sloop_core.cmake)
    include(${COMPONENT_DIR}/../../cmake/sloop_core.cmake)
    set(EDITOR_DIR "${CMAKE_CURRENT_BINARY_DIR}/editor")
    sloop_editor_prepare("${EDITOR_DIR}")
    file(GLOB WEB_FILES CONFIGURE_DEPENDS "${WEB_DIR}/*.html" "${WEB_DIR}/*.js" "${WEB_DIR}/*.css")
    list(SORT WEB_FILES)
    list(APPEND WEB_FILES "${EDITOR_DIR}/editor.html" "${EDITOR_DIR}/fukiai.ttf" "${EDITOR_DIR}/FUKIAI-LICENSE.txt")
    idf_build_get_property(python PYTHON)
    set(decl "")
    set(table "")
    foreach(path IN LISTS WEB_FILES)
        get_filename_component(f "${path}" NAME)
        string(MAKE_C_IDENTIFIER "web_${f}" sym)
        set(gz "${CMAKE_CURRENT_BINARY_DIR}/web_gz/${f}.gz")
        add_custom_command(OUTPUT "${gz}"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/web_gz"
            COMMAND ${python} -c "import gzip,sys; open(sys.argv[2],'wb').write(gzip.compress(open(sys.argv[1],'rb').read(),9,mtime=0))" "${path}" "${gz}"
            DEPENDS "${path}" VERBATIM)
        target_add_binary_data(${COMPONENT_LIB} "${gz}" BINARY RENAME_TO ${sym} DEPENDS "${gz}")
        if(f MATCHES "\\.html$")
            set(type "text/html; charset=utf-8")
        elseif(f MATCHES "\\.css$")
            set(type "text/css; charset=utf-8")
        elseif(f MATCHES "\\.ttf$")
            set(type "font/ttf")
        elseif(f MATCHES "\\.txt$")
            set(type "text/plain; charset=utf-8")
        else()
            set(type "text/javascript; charset=utf-8")
        endif()
        string(APPEND decl "extern const uint8_t _binary_${sym}_start[], _binary_${sym}_end[];\n")
        string(APPEND table "    {\"/${f}\", _binary_${sym}_start, _binary_${sym}_end, \"${type}\"},\n")
    endforeach()
    list(LENGTH WEB_FILES n)
    file(CONFIGURE OUTPUT "${GEN}" CONTENT "/* generated by components/web_server/CMakeLists.txt from web/ */
#include \"web_assets.h\"
${decl}
const web_asset_t web_assets[] = {
${table}};
const size_t web_assets_count = ${n};
")
endif()
